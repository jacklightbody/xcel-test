Example Test Cases

Sample test cases demonstrating the new relative reference functionality.

Complete Example

[
  {
    "name": "Relative Reference Example",
    "inputs": [
      {
        "cell": "Sheet1!B2",
        "value": 0.05
      },
      {
        "relativeTo": {
          "sheet": "Sheet1",
          "referenceCell": "Total Income",
          "colOffset": 2,
          "rowOffset": -1
        },
        "value": 100000
      }
    ],
    "assertions": [
      {
        "cell": "Sheet1!E12",
        "equals": 1234567,
        "tolerance": 1
      },
      {
        "relativeTo": {
          "sheet": "Sheet1",
          "referenceColCell": "Total Income",
          "referenceRowCell": "Last fiscal year"
        },
        "equals": 2469134,
        "tolerance": 1
      }
    ]
  }
]
